#!/usr/bin/env bash
set -euo pipefail

# папки и файлы 

INPUT_DIR="reads"             # папка с исходными fastq
TRIMMED_DIR="trimmed_fastq"   # папка для очищенных данных
FASTQC_DIR="fastqc_trimmed"   # папка для FastQC
GENOME="ha38.fa"  # файл генома

# обрезка адаптеров
for i in {19..24}; do
  cutadapt \
    -a AGATCGGAAGAGC \
    -A AGATCGGAAGAGC \
    -o "$OUTPUT_DIR/${i}_R1_cutadapt.fq.gz" \
    -p "$OUTPUT_DIR/${i}_R2_cutadapt.fq.gz" \
    "$INPUT_DIR/${i}_S${i}_L001_R1_001.fastq.gz" \
    "$INPUT_DIR/${i}_S${i}_L001_R2_001.fastq.gz" \
    -j 4 \
    -q 20 \
    --minimum-length 35
done

# FastQC после обрезки адаптеров

fastqc "$TRIMMED_DIR"/*_paired.fastq.gz \
       -o "$FASTQC_DIR" \
       -t 4 \
       --quiet

# индексация генома (разовая)

bwa index "$GENOME"

# выравнивание BWA + сортировка

for i in {19..24}; do
  bwa mem -t 8 -M "$GENOME" \
    "$TRIMMED_DIR/${i}_S${i}_L001_R1_paired.fastq.gz" \
    "$TRIMMED_DIR/${i}_S${i}_L001_R2_paired.fastq.gz" \
  | samtools view -b - \
  | samtools sort -@ 4 -o "${i}.sorted.bam"
done

# индексация BAM

for i in {19..24}; do
  samtools index "${i}.sorted.bam"
done

# покрытие 
for i in {19..24}; do

  # покрытие генома
  bedtools genomecov -ibam "${i}.sorted.bam" -bga > "${i}.cov.bedgraph"

  # глубинный фильтр
  awk '$4>0' "${i}.cov.bedgraph" > "${i}.cov.tsv"

  # мердж
  bedtools merge -i "${i}.cov.tsv" > "${i}.regions.bed"

  # лдина мерджа
  awk '{print $1"\t"$2"\t"$3"\t"($3-$2)}' "${i}.regions.bed" > "${i}.lengths.tsv"

  # бедтулс для регионов
  bedtools sort -i "${i}.regions.bed" > "${i}.regions_sorted.bed"
  bedtools closest -a "${i}.regions_sorted.bed" -b "${i}.regions_sorted.bed" -d \
    | awk '$1==$4 {print $1, $10}' > "${i}.distances.tsv"

done
